Structured Outputs & Multi-Turn Chat in Go

By configuring model.ResponseSchema using Go SDK's *genai.Schema alongside model.ResponseMIMEType = "application/json", the Gemini model is bound to valid JSON representations matching your structural grammar across continuous chat.SendMessage(ctx, ...) session dialogues.

package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"os"

	"github.com/google/generative-ai-go/genai"
	"google.golang.org/api/option"
)

func main() {
	ctx := context.Background()
	apiKey := os.Getenv("GEMINI_API_KEY")

	// Initialize Generative AI client
	client, err := genai.NewClient(ctx, option.WithAPIKey(apiKey))
	if err != nil {
		log.Fatalf("Failed to create client: %v", err)
	}
	defer client.Close()

	model := client.GenerativeModel("gemini-2.5-flash")

	// Step 1: System Instruction ensures agent behavioral constraints
	model.SystemInstruction = genai.NewUserContent(
		genai.Text("You are an architectural state manager. Always respond in valid JSON matching schema."),
	)

	// Step 2: Enforce strict application/json and define OpenAPI Schema
	model.ResponseMIMEType = "application/json"
	model.ResponseSchema = &genai.Schema{
		Type: genai.TypeObject,
		Properties: map[string]*genai.Schema{
			"project_name": {
				Type:        genai.TypeString,
				Description: "Canonical identifier for the active initiative.",
			},
			"priority": {
				Type:        genai.TypeString,
				Enum:        []string{"Low", "Medium", "High"},
				Description: "Calculated escalation tier.",
			},
			"tags": {
				Type:        genai.TypeArray,
				Description: "Technological component stack tags.",
				Items: &genai.Schema{
					Type: genai.TypeString,
				},
			},
		},
		Required: []string{"project_name", "priority", "tags"},
	}

	// Step 3: Initiate multi-turn stateful chat session
	chat := model.StartChat()

	// Turn 1: Initial creation request
	resp1, err := chat.SendMessage(ctx, genai.Text("Initialize Project Hydro with High priority using Go and gRPC"))
	if err != nil {
		log.Fatalf("Turn 1 error: %v", err)
	}
	fmt.Println("--- Turn 1 Output ---")
	printFirstPart(resp1)

	// Turn 2: Subsequent request modifying accumulated chat context
	resp2, err := chat.SendMessage(ctx, genai.Text("Please add Kubernetes to the stack"))
	if err != nil {
		log.Fatalf("Turn 2 error: %v", err)
	}
	fmt.Println("--- Turn 2 Output ---")
	printFirstPart(resp2)
}

func printFirstPart(resp *genai.GenerateContentResponse) {
	for _, cand := range resp.Candidates {
		for _, part := range cand.Content.Parts {
			fmt.Printf("%s\n", part)
		}
	}
}
