package main

import (
    "context"
    "errors"
    "fmt"
    "io"
    "log"
    "os"
    "sync"

    "google.golang.org/api/iterator"
    "github.com/google/generative-ai-go/genai"
)

// SessionManager guarantees thread-safe multi-user access to active Gemini chat sessions.
type SessionManager struct {
    mu       sync.RWMutex
    model    *genai.GenerativeModel
    sessions map[string]*genai.ChatSession
}

func NewSessionManager(model *genai.GenerativeModel) *SessionManager {
    return &SessionManager{
        model:    model,
        sessions: make(map[string]*genai.ChatSession),
    }
}

// GetOrCreateSession safely retrieves or initializes a ChatSession for a given user.
func (sm *SessionManager) GetOrCreateSession(userID string) *genai.ChatSession {
    sm.mu.Lock()
    defer sm.mu.Unlock()

    if session, exists := sm.sessions[userID]; exists {
        return session
    }

    session := sm.model.StartChat()
    sm.sessions[userID] = session
    return session
}

// StreamUserMessage issues a streaming prompt and pipes token chunks directly to an io.Writer.
func (sm *SessionManager) StreamUserMessage(
    ctx context.Context, 
    userID string, 
    prompt string, 
    out io.Writer,
) error {
    session := sm.GetOrCreateSession(userID)

    // SendMessageStream returns a non-blocking iterator
    iter := session.SendMessageStream(ctx, genai.Text(prompt))

    for {
        resp, err := iter.Next()
        if errors.Is(err, iterator.Done) {
            break
        }
        if err != nil {
            return fmt.Errorf("stream iteration error: %w", err)
        }

        for _, cand := range resp.Candidates {
            if cand.Content != nil {
                for _, part := range cand.Content.Parts {
                    fmt.Fprintf(out, "[User: %s] %v\n", userID, part)
                }
            }
        }
    }
    return nil
}

func main() {
    ctx := context.Background()
    client, err := genai.NewClient(ctx)
    if err != nil {
        log.Fatalf("Failed to create genai client: %v", err)
    }
    defer client.Close()

    model := client.GenerativeModel("gemini-2.5-flash")
    sm := NewSessionManager(model)

    var wg sync.WaitGroup

    // Goroutine 1: user_42 stream
    wg.Add(1)
    go func() {
        defer wg.Done()
        _ = sm.StreamUserMessage(ctx, "user_42", "Tell me a 1-sentence tip on writing clean Go", os.Stdout)
    }()

    // Goroutine 2: user_99 stream
    wg.Add(1)
    go func() {
        defer wg.Done()
        _ = sm.StreamUserMessage(ctx, "user_99", "Give me a single Go interface example", os.Stdout)
    }()

    wg.Wait()
}

Concurrency API & Type Signatures

google.golang.org/genai
sync.RWMutex stdlib

Guards the user sessions map. Safe for multiple concurrent readers using sm.mu.RLock() and exclusive writes using sm.mu.Lock().

SendMessageStream genai

Spawns an HTTP streaming session returning a *GenerateContentResponseIterator. Each iteration handles independent tokens safely.

errors.Is(iterator.Done) iterator

Graceful sentinel check signaling end-of-stream. Prevents goroutines from hanging and allows safe channel teardowns.
